"""研究协议的样例与反例检查；完整载入仅用于宿主核验，不是 App 流式导入器。

依赖 jsonschema 4.23.0；使用 PYTHONPATH 指向隔离安装目录运行本文件。
不连接任何应用数据库，不保存一万任务的完整样本到仓库。
"""
import copy
import json
import sys
import time
from datetime import date, datetime, time as walltime, timedelta, timezone
from pathlib import Path
from zoneinfo import ZoneInfo

from jsonschema import Draft202012Validator, FormatChecker

ROOT = Path(__file__).resolve().parents[1]
SCHEMA = json.loads((ROOT / "mission-book.goal-package.v1.schema.json").read_text())
Draft202012Validator.check_schema(SCHEMA)
VALIDATOR = Draft202012Validator(SCHEMA, format_checker=FormatChecker())
NOW = datetime(2026, 10, 27, 4, tzinfo=timezone.utc)
MAX_BYTES = 32 * 1024 * 1024


class Invalid(ValueError):
    pass


def pairs(items):
    result = {}
    for key, value in items:
        if key in result:
            raise Invalid("DUPLICATE_FIELD")
        result[key] = value
    return result


def encode(value):
    return json.dumps(value, ensure_ascii=False, separators=(",", ":")).encode()


def location(name):
    try:
        return ZoneInfo(name)
    except (KeyError, ValueError) as error:
        raise Invalid("UNKNOWN_TIME_ZONE") from error


def instant(text, zone):
    try:
        value = datetime.fromisoformat(text.replace("Z", "+00:00"))
    except ValueError as error:
        raise Invalid("INVALID_DATE") from error
    local = value.astimezone(zone)
    if not text.endswith("Z"):
        if (value.replace(tzinfo=None) != local.replace(tzinfo=None)
                or value.utcoffset() != local.utcoffset()):
            raise Invalid("OFFSET_ZONE_MISMATCH")
    return value.astimezone(timezone.utc)


def midnight(day, zone):
    naive = datetime.combine(day, walltime())
    early = naive.replace(tzinfo=zone, fold=0)
    late = naive.replace(tzinfo=zone, fold=1)
    if early.astimezone(timezone.utc).astimezone(zone).replace(tzinfo=None) != naive:
        raise Invalid("NONEXISTENT_MIDNIGHT")
    if early.utcoffset() != late.utcoffset():
        raise Invalid("AMBIGUOUS_MIDNIGHT")
    return early.astimezone(timezone.utc)


def check(raw):
    if len(raw) > MAX_BYTES:
        raise Invalid("LIMIT_BYTES")
    if raw.startswith(b"\xef\xbb\xbf"):
        raise Invalid("BOM")
    value = json.loads(raw.decode("utf-8"), object_pairs_hook=pairs)
    stack = [(value, 1)]
    values = 0
    while stack:
        item, depth = stack.pop()
        values += 1
        if depth > 64 or values > 1000000:
            raise Invalid("LIMIT_JSON_STRUCTURE")
        if isinstance(item, dict):
            stack.extend((part, depth + 1) for part in item.values())
        elif isinstance(item, list):
            stack.extend((part, depth + 1) for part in item)
    if next(VALIDATOR.iter_errors(value), None) is not None:
        raise Invalid("SCHEMA")
    if type(value["version"]) is not int:
        raise Invalid("INTEGER_TYPE")
    defaults = value["defaults"]
    if type(defaults["plannedDurationMinutes"]) is not int:
        raise Invalid("INTEGER_TYPE")
    location(defaults["timeZone"])
    keys = set()
    node_count = 0
    stack = [(node, 1) for node in value["nodes"]]
    while stack:
        node, depth = stack.pop()
        node_count += 1
        if depth > 12 or node_count > 500:
            raise Invalid("LIMIT_NODE_TREE")
        if node["key"] in keys:
            raise Invalid("DUPLICATE_NODE_KEY")
        keys.add(node["key"])
        stack.extend((child, depth + 1) for child in node.get("children", []))
    mission_keys = set()
    steps_count = 0
    resolved = []
    for mission in value["missions"]:
        if len(encode(mission)) > 256 * 1024:
            raise Invalid("LIMIT_MISSION_BYTES")
        if mission["key"] in mission_keys:
            raise Invalid("DUPLICATE_MISSION_KEY")
        mission_keys.add(mission["key"])
        if "nodeKey" in mission and mission["nodeKey"] not in keys:
            raise Invalid("UNKNOWN_NODE_KEY")
        plan = mission.get("executionPlan", {})
        steps_count += len(plan.get("steps", []))
        if steps_count > 100000:
            raise Invalid("LIMIT_TOTAL_STEPS")
        schedule = mission["schedule"]
        zone = location(schedule.get("timeZone", defaults["timeZone"]))
        day = date.fromisoformat(schedule["scheduledDate"])
        duration = schedule.get("plannedDurationMinutes", defaults["plannedDurationMinutes"])
        if type(duration) is not int:
            raise Invalid("INTEGER_TYPE")
        start = None
        if schedule["timePolicy"] == "FIXED":
            start = instant(schedule["fixedStartAt"], zone)
            if start.astimezone(zone).date() != day:
                raise Invalid("START_DATE_MISMATCH")
        policy = mission.get("verificationPolicy", defaults["verificationPolicy"])
        deadline = None
        if policy == "REQUIRED_EVIDENCE":
            if not plan.get("verificationCriteria", "").strip():
                raise Invalid("REQUIRED_CRITERIA")
            deadline = (instant(schedule["evidenceDeadlineAt"], zone)
                        if "evidenceDeadlineAt" in schedule
                        else midnight(day + timedelta(days=1), zone))
            lower = start if start is not None else midnight(day, zone)
            if deadline <= NOW or deadline <= lower:
                raise Invalid("INVALID_DEADLINE")
        elif "evidenceDeadlineAt" in schedule:
            raise Invalid("UNEXPECTED_DEADLINE")
        resolved.append({
            "key": mission["key"],
            "duration": duration,
            "policy": policy,
            "deadline": deadline.isoformat().replace("+00:00", "Z") if deadline else None,
        })
    return {"nodes": node_count, "missions": len(resolved),
            "totalMinutes": sum(row["duration"] for row in resolved), "resolved": resolved}


SAMPLE = json.loads((ROOT / "examples/课标试运行.goal-package.v1.json").read_text())
results = []


def case(name, value, allowed, raw=False):
    try:
        result = check(value if raw else encode(value))
        accepted, reason = True, "ACCEPTED"
    except (ValueError, UnicodeError) as error:
        accepted, reason = False, str(error).splitlines()[0][:150]
        result = None
    results.append({"case": name, "expectedAccepted": allowed,
                    "accepted": accepted, "matched": accepted == allowed, "reason": reason})
    return result


def change(path, value, delete=False):
    item = copy.deepcopy(SAMPLE)
    target = item
    for part in path[:-1]:
        target = target[part]
    if delete:
        del target[path[-1]]
    else:
        target[path[-1]] = value
    return item
