class _Counter extends QueryInterceptor {
  int selects = 0;
  @override
  Future<List<Map<String, Object?>>> runSelect(
    QueryExecutor executor,
    String statement,
    List<Object?> args,
  ) {
    selects++;
    return super.runSelect(executor, statement, args);
  }
}

class _MemoryStore implements EvidenceImageStore {
  final files = <String, Uint8List>{};
  int reads = 0;
  @override
  Future<void> writeNew(String key, Uint8List bytes) async =>
      files[key] = bytes;
  @override
  Future<Uint8List> read(String key) async {
    reads++;
    return files[key] ?? (throw StateError('missing $key'));
  }

  @override
  Future<void> delete(String key) async => files.remove(key);
  @override
  Future<Set<String>> listKeys() async => files.keys.toSet();
}

class _Picker implements EvidenceImagePicker, EvidenceFilePicker {
  @override
  Future<SelectedEvidenceImage?> pick() async => SelectedEvidenceImage(
    fileName: 'note.txt',
    byteLength: 3,
    readBytes: () async => Uint8List.fromList([1, 2, 3]),
  );
}

class _ControlledGoals extends GoalService {
  _ControlledGoals(super.repository);
  Completer<Goal?>? nextRead;
  @override
  Future<Goal?> findById(String id) => nextRead?.future ?? super.findById(id);
}

  late DriftMissionRepository repo;
  late _Counter counter;
  final now = DateTime.utc(2026, 10, 27, 6);
  setUp(() async {
    counter = _Counter();
    db = AppDatabase(executor: NativeDatabase.memory().interceptWith(counter));
    repo = DriftMissionRepository(db, clock: () => now);
    missions = MissionService(repo, clock: () => now);
    goals = GoalService(DriftGoalRepository(db));

    await missions.dispose();
    await goals.dispose();
    await db.close();
  });

    title: title,
    scheduledDate: '2026-10-27',

    tester.view.devicePixelRatio = 1;
    addTearDown(tester.view.resetPhysicalSize);
    addTearDown(tester.view.resetDevicePixelRatio);
    await tester.pumpWidget(
      MaterialApp(
        theme: AppTheme.light,

        supportedLocales: const [Locale('zh', 'CN')],
        localizationsDelegates: GlobalMaterialLocalizations.delegates,
        home: child,
      ),

    await tester.pumpAndSettle();
  }

  Future<void> tapKey(WidgetTester tester, String key) async {
    final target = find.byKey(Key(key));
    expect(target, findsOneWidget);
    await tester.ensureVisible(target);
    await tester.pumpAndSettle();
    await tester.tap(target);
    await tester.pumpAndSettle();
  }

  test('AUDIT-09 备份等待附件 IO 期间不应占用数据库操作队列', () async {
    await missions.create(
      title: '备份附件',
      scheduledDate: '2026-10-27',
      timeZone: 'Asia/Shanghai',
      plannedDurationMinutes: 10,
      verificationPolicy: VerificationPolicy.optionalEvidence,
    );
    var detail = (await repo.findAllDetails()).single;
    await missions.execute(
      missions.prepareExecution(detail: detail, type: ExecutionEventType.start),
    );
    detail = (await repo.findDetail(detail.mission.id))!;
    await missions.execute(
      missions.prepareExecution(
        detail: detail,
        type: ExecutionEventType.end,
        report: UserReportedResult.completed,
        submitOptionalEvidence: true,
      ),
    );
    await repo.submitEvidence(
      EvidenceSubmission(
        id: 'batch-a',
        missionId: detail.mission.id,
        items: [
          EvidenceDraft(
            id: 'file-a',
            type: EvidenceType.file,
            source: EvidenceSource.fileSystem,
            attachmentStorageKey: 'attachments/${detail.mission.id}/file-a.txt',
            attachmentFileName: 'file.txt',
          ),
        ],
      ),
    );
    await createTask('等待取消');
    final target = (await repo.findAllDetails()).firstWhere(
      (d) => d.mission.title == '等待取消',
    );
    final reading = Completer<void>();
    final release = Completer<Uint8List>();
    final exporter = DatabaseBackupService(
      database: db,
      readAttachment: (_) {
        reading.complete();
        return release.future;
      },
    );
    final backup = exporter.create();
    await reading.future;
    var cancelled = false;
    final cancel = missions
        .cancel(missionId: target.mission.id)
        .then((_) => cancelled = true);
    await Future<void>.delayed(const Duration(milliseconds: 50));
    final duringRead = cancelled;
    release.complete(Uint8List.fromList([1, 2, 3]));
    await backup;
    await cancel;
    print(
      'AUDIT BACKUP cancelFinishedWhileReading=$duringRead after=${(await repo.findDetail(target.mission.id))!.phase}',
    );
    expect(duringRead, isTrue);
  });

  testWidgets('AUDIT-11 尚未打开历史页面时不应读取全部历史原图', (tester) async {
    final recorder = ui.PictureRecorder();
    ui.Canvas(recorder).drawColor(const ui.Color(0xff244d3d), ui.BlendMode.src);
    final picture = recorder.endRecording();
    final png = await tester.runAsync(() async {
      final image = await picture.toImage(2, 2);
      final bytes = (await image.toByteData(
        format: ui.ImageByteFormat.png,
      ))!.buffer.asUint8List();
      image.dispose();
      return bytes;
    });
    picture.dispose();
    final store = _MemoryStore();
    await missions.dispose();
    missions = MissionService(
      repo,
      clock: () => now,
      images: EvidenceImageService(store: store, picker: _Picker()),
    );
    for (var i = 0; i < 20; i++) {
      final request = missions.prepareCreation(
        title: '历史图片-$i',
        scheduledDate: '2026-10-26',
        timeZone: 'Asia/Shanghai',
        plannedDurationMinutes: 10,
        verificationPolicy: VerificationPolicy.optionalEvidence,
      );
      await missions.createPrepared(request);
      var detail = (await repo.findDetail(request.mission.id))!;
      await missions.execute(
        missions.prepareExecution(
          detail: detail,
          type: ExecutionEventType.start,
        ),
      );
      detail = (await repo.findDetail(detail.mission.id))!;
      await missions.execute(
        missions.prepareExecution(
          detail: detail,
          type: ExecutionEventType.end,
          report: UserReportedResult.completed,
          submitOptionalEvidence: true,
        ),
      );
      final key = 'attachments/${detail.mission.id}/image-$i.png';
      store.files[key] = png!;
      await repo.submitEvidence(
        EvidenceSubmission(
          id: 'batch-$i',
          missionId: detail.mission.id,
          items: [
            EvidenceDraft(
              id: 'image-$i',
              type: EvidenceType.image,
              source: EvidenceSource.fileSystem,
              attachmentStorageKey: key,
              attachmentFileName: '$i.png',
            ),
          ],
        ),
      );
    }
    await mount(
      tester,
      AppShell(
        missionService: missions,
        goalService: goals,
        backupService: TestBackup(),
        device: TestAppDevice(),
        clock: () => now,
      ),
    );
    final reads = store.reads;
    await tester.pumpWidget(const SizedBox.shrink());
    print('AUDIT HIDDEN HISTORY imageReads=$reads total=20 selected=today');
    expect(reads, 0);
  });
